-- =====================================================================
--  SPORTSHUB | MULTI-SPORT MANAGEMENT & FACILITY BOOKING PLATFORM
-- =====================================================================
--
-- Project:
--     SportsHub
--
-- Purpose:
--     A simplified but complete multi-sport platform where users can:
--       1. View matches, teams, players, statistics and tournaments
--       2. Create/manage teams
--       3. Find nearby opponent teams
--       4. Book nearby grounds, turfs and courts
--       5. Organize tournaments
--       6. Perform authorized live scoring
--       7. Discover sports facilities using location
--       8. Receive notifications and review facilities
--
-- Main Users:
--     VIEWER -> views SportsHub information
--     PLAYER -> creates/manages teams and participates in matches
--     OWNER  -> manages facilities and tournaments
--     ADMIN  -> system administrator
--
-- Sports in V1:
--     Cricket, Football, Basketball, Badminton, Tennis, Volleyball
--
-- RDBMS:
--     MySQL 8.0+
--
-- IMPORTANT DESIGN DECISION:
--     The original design had many small tables. This version deliberately
--     reduces the database to 13 tables by merging tables where a separate
--     table is not essential for the learning project.
--
--     Merged/reduced areas:
--       * Facility images + slots -> facilities
--       * Tournament registrations -> JSON inside tournaments
--       * Match requests + scorer assignment -> matches
--       * All sport events -> match_events
--       * Payments -> bookings
--       * Admin audit/favorites are optional and omitted from V1
--
--     This keeps the project much easier to understand while still covering
--     the required SportsHub features.
--

-- =====================================================================
-- PART 1 : DATABASE CREATION
-- =====================================================================

-- Why?
-- Creates a clean database so the complete script can be executed from
-- top to bottom during development.
-- WARNING: DROP DATABASE permanently removes the old database.

DROP DATABASE IF EXISTS sportshub;
CREATE DATABASE sportshub;
USE sportshub;


-- =====================================================================
-- PART 2 : CORE TABLES
-- =====================================================================

-- ---------------------------------------------------------------------
-- 2.1 USERS
-- ---------------------------------------------------------------------
-- Why?
-- Stores login/account information for every SportsHub user.
--
-- role:
--   VIEWER -> mainly views public SportsHub information
--   PLAYER -> creates/manages teams and participates in matches
--   OWNER  -> manages facilities and tournaments
--   ADMIN  -> manages the platform
--
-- Location is stored here because the application can use the user's
-- city/coordinates as the starting point for nearby searches.
--
-- password_hash must contain a real hash generated by Flask/Python.
-- Never store a plain-text password in production.
--
-- This learning version uses one primary role per account. If later you
-- want one account to be both PLAYER and OWNER, a user_roles table can be
-- introduced without changing the rest of the design.
-- ---------------------------------------------------------------------
CREATE TABLE users (
    user_id       INT AUTO_INCREMENT PRIMARY KEY,
    full_name     VARCHAR(100) NOT NULL,
    email         VARCHAR(100) NOT NULL UNIQUE,
    phone         VARCHAR(15) UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    role          ENUM('VIEWER','PLAYER','OWNER','ADMIN') NOT NULL,
    city          VARCHAR(100),
    latitude      DECIMAL(10,8),
    longitude     DECIMAL(11,8),
    is_active     BOOLEAN DEFAULT TRUE,
    created_at    DATETIME DEFAULT CURRENT_TIMESTAMP
);


-- ---------------------------------------------------------------------
-- 2.2 SPORTS
-- ---------------------------------------------------------------------
-- Why?
-- Stores the master list of sports instead of repeatedly writing sport
-- names in every table.
--
-- Teams, facilities, tournaments and matches point to this table.
-- This makes the application genuinely multi-sport and easy to extend.
-- ---------------------------------------------------------------------
CREATE TABLE sports (
    sport_id      INT AUTO_INCREMENT PRIMARY KEY,
    sport_name    VARCHAR(50) NOT NULL UNIQUE,
    description   VARCHAR(255),
    is_active     BOOLEAN DEFAULT TRUE
);


-- ---------------------------------------------------------------------
-- 2.3 PLAYER PROFILES
-- ---------------------------------------------------------------------
-- Why?
-- users contains account/login information.
-- player_profiles contains optional information specifically related to
-- a player.
--
-- Keeping this separate means a VIEWER or OWNER does not need player-only
-- columns filled in.
-- ---------------------------------------------------------------------
CREATE TABLE player_profiles (
    player_id      INT AUTO_INCREMENT PRIMARY KEY,
    user_id        INT NOT NULL UNIQUE,
    date_of_birth  DATE,
    gender         VARCHAR(20),
    bio            VARCHAR(500),
    experience     INT DEFAULT 0,
    FOREIGN KEY (user_id) REFERENCES users(user_id) ON DELETE CASCADE
);


-- =====================================================================
-- PART 3 : TEAMS & PLAYERS
-- =====================================================================

-- ---------------------------------------------------------------------
-- 3.1 TEAMS
-- ---------------------------------------------------------------------
-- Why?
-- Stores teams created by PLAYER users.
--
-- One player can create teams for different sports:
--   Royal Strikers -> Cricket
--   Vizag FC       -> Football
--   Hoop Warriors  -> Basketball
--
-- latitude/longitude make the team searchable for the Find Opponent
-- feature.
--
-- The creator is linked to users, while sport_id is linked to sports.
-- ---------------------------------------------------------------------
CREATE TABLE teams (
    team_id       INT AUTO_INCREMENT PRIMARY KEY,
    team_name     VARCHAR(100) NOT NULL,
    sport_id      INT NOT NULL,
    created_by    INT NOT NULL,
    city          VARCHAR(100),
    latitude      DECIMAL(10,8),
    longitude     DECIMAL(11,8),
    description   VARCHAR(500),
    is_active     BOOLEAN DEFAULT TRUE,
    created_at    DATETIME DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(team_name, sport_id),
    FOREIGN KEY (sport_id) REFERENCES sports(sport_id),
    FOREIGN KEY (created_by) REFERENCES users(user_id)
);


-- ---------------------------------------------------------------------
-- 3.2 TEAM MEMBERS
-- ---------------------------------------------------------------------
-- Why?
-- A team has many players and a player can belong to many teams.
-- This is a many-to-many relationship, so a bridge table is necessary.
--
-- member_role explains the person's responsibility inside the team.
-- CAPTAIN and SCORER can later be used by the Flask authorization layer.
-- ---------------------------------------------------------------------
CREATE TABLE team_members (
    team_member_id INT AUTO_INCREMENT PRIMARY KEY,
    team_id        INT NOT NULL,
    user_id        INT NOT NULL,
    member_role    ENUM('PLAYER','CAPTAIN','SCORER') DEFAULT 'PLAYER',
    joined_at      DATETIME DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(team_id, user_id),
    FOREIGN KEY (team_id) REFERENCES teams(team_id) ON DELETE CASCADE,
    FOREIGN KEY (user_id) REFERENCES users(user_id) ON DELETE CASCADE
);


-- =====================================================================
-- PART 4 : FACILITIES & LOCATIONS
-- =====================================================================

-- ---------------------------------------------------------------------
-- 4.1 FACILITIES
-- ---------------------------------------------------------------------
-- Why?
-- Stores sports grounds, turfs and courts owned by OWNER users.
--
-- To reduce tables, image_url and basic availability are stored directly
-- in this table instead of creating facility_images and facility_slots.
--
-- Example:
--   Cricket  -> Cricket Ground
--   Football -> Football Turf
--   Tennis   -> Tennis Court
--
-- available_from / available_to describe the normal operating window.
-- The application can generate hourly slots from these values.
-- ---------------------------------------------------------------------
CREATE TABLE facilities (
    facility_id     INT AUTO_INCREMENT PRIMARY KEY,
    facility_name   VARCHAR(150) NOT NULL,
    owner_id        INT NOT NULL,
    sport_id        INT NOT NULL,
    facility_type   VARCHAR(50) NOT NULL,
    description     VARCHAR(500),
    city            VARCHAR(100),
    address         VARCHAR(255),
    latitude        DECIMAL(10,8),
    longitude       DECIMAL(11,8),
    image_url       VARCHAR(500),
    price_per_hour  DECIMAL(10,2) NOT NULL,
    available_from  TIME DEFAULT '06:00:00',
    available_to    TIME DEFAULT '22:00:00',
    approval_status ENUM('PENDING','APPROVED','REJECTED') DEFAULT 'PENDING',
    is_active       BOOLEAN DEFAULT TRUE,
    created_at      DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (owner_id) REFERENCES users(user_id),
    FOREIGN KEY (sport_id) REFERENCES sports(sport_id)
);


-- =====================================================================
-- PART 5 : TOURNAMENTS
-- =====================================================================

-- ---------------------------------------------------------------------
-- 5.1 TOURNAMENTS
-- ---------------------------------------------------------------------
-- Why?
-- Stores tournaments organized by OWNER users.
--
-- To reduce the number of tables, participating team IDs are stored in a
-- JSON array instead of using a separate tournament_teams table.
--
-- Example:
--   registered_team_ids = [1,2,5]
--
-- The backend can update this JSON array when a team registers.
-- For a very large production system, a separate tournament_teams table
-- would be preferable, but JSON keeps this student project simpler.
-- ---------------------------------------------------------------------
CREATE TABLE tournaments (
    tournament_id       INT AUTO_INCREMENT PRIMARY KEY,
    tournament_name     VARCHAR(150) NOT NULL,
    sport_id            INT NOT NULL,
    organizer_id        INT NOT NULL,
    city                VARCHAR(100),
    venue               VARCHAR(255),
    latitude            DECIMAL(10,8),
    longitude           DECIMAL(11,8),
    start_date          DATE NOT NULL,
    end_date            DATE NOT NULL,
    max_teams           INT DEFAULT 16,
    registration_fee    DECIMAL(10,2) DEFAULT 0.00,
    registered_team_ids JSON,
    status              ENUM('UPCOMING','OPEN','ONGOING','COMPLETED','CANCELLED') DEFAULT 'UPCOMING',
    created_at          DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (sport_id) REFERENCES sports(sport_id),
    FOREIGN KEY (organizer_id) REFERENCES users(user_id)
);


-- =====================================================================
-- PART 6 : MATCHES & FIND OPPONENT
-- =====================================================================

-- ---------------------------------------------------------------------
-- 6.1 MATCHES
-- ---------------------------------------------------------------------
-- Why?
-- This is the central table for actual matches.
--
-- It connects:
--   sport
--   team A
--   team B
--   optional tournament
--   optional facility
--   creator
--   date/time
--   winner
--
-- To reduce tables, this table also stores Find Opponent request data
-- and the authorized scorer.
--
-- match_type:
--   NORMAL     -> directly scheduled match
--   OPPONENT   -> created through Find Opponent
--   TOURNAMENT -> tournament match
--
-- request_status is used only for OPPONENT matches.
-- A NULL target team means the request is open to suitable nearby teams.
-- ---------------------------------------------------------------------
CREATE TABLE matches (
    match_id            INT AUTO_INCREMENT PRIMARY KEY,
    sport_id            INT NOT NULL,
    team_a_id           INT NOT NULL,
    team_b_id           INT NULL,
    tournament_id       INT NULL,
    facility_id         INT NULL,
    created_by          INT NOT NULL,
    match_date          DATE NOT NULL,
    start_time          TIME NOT NULL,
    match_type          ENUM('NORMAL','OPPONENT','TOURNAMENT') DEFAULT 'NORMAL',
    request_status      ENUM('NONE','OPEN','PENDING','ACCEPTED','REJECTED','CANCELLED','EXPIRED') DEFAULT 'NONE',
    search_latitude     DECIMAL(10,8),
    search_longitude    DECIMAL(11,8),
    search_radius_km    DECIMAL(5,2) DEFAULT 10.00,
    request_message     VARCHAR(500),
    scorer_id           INT NULL,
    status              ENUM('SCHEDULED','LIVE','COMPLETED','CANCELLED') DEFAULT 'SCHEDULED',
    winner_team_id      INT NULL,
    score_a             INT DEFAULT 0,
    score_b             INT DEFAULT 0,
    created_at          DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (sport_id) REFERENCES sports(sport_id),
    FOREIGN KEY (team_a_id) REFERENCES teams(team_id),
    FOREIGN KEY (team_b_id) REFERENCES teams(team_id),
    FOREIGN KEY (tournament_id) REFERENCES tournaments(tournament_id),
    FOREIGN KEY (facility_id) REFERENCES facilities(facility_id),
    FOREIGN KEY (created_by) REFERENCES users(user_id),
    FOREIGN KEY (scorer_id) REFERENCES users(user_id),
    FOREIGN KEY (winner_team_id) REFERENCES teams(team_id)
);


-- =====================================================================
-- PART 7 : BOOKINGS & PAYMENTS
-- =====================================================================

-- ---------------------------------------------------------------------
-- 7.1 BOOKINGS
-- ---------------------------------------------------------------------
-- Why?
-- Stores a user's facility reservation.
--
-- To reduce tables, payment information is stored here instead of using
-- a separate payments table.
--
-- Security rule:
--   Never store card number, CVV, UPI PIN or bank password.
--   transaction_reference is only a reference returned by a payment
--   gateway or a demo transaction ID.
--
-- The backend should check overlapping bookings before confirmation.
-- ---------------------------------------------------------------------
CREATE TABLE bookings (
    booking_id            INT AUTO_INCREMENT PRIMARY KEY,
    user_id               INT NOT NULL,
    facility_id           INT NOT NULL,
    booking_date          DATE NOT NULL,
    start_time            TIME NOT NULL,
    end_time              TIME NOT NULL,
    amount                DECIMAL(10,2) NOT NULL,
    booking_status        ENUM('PENDING','CONFIRMED','CANCELLED','COMPLETED') DEFAULT 'PENDING',
    payment_method        ENUM('UPI','CARD','CASH','NONE') DEFAULT 'NONE',
    payment_status        ENUM('PENDING','PAID','FAILED','REFUNDED') DEFAULT 'PENDING',
    transaction_reference VARCHAR(100),
    created_at            DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(user_id),
    FOREIGN KEY (facility_id) REFERENCES facilities(facility_id)
);


-- =====================================================================
-- PART 8 : COMMON LIVE SCORING
-- =====================================================================

-- ---------------------------------------------------------------------
-- 8.1 MATCH EVENTS
-- ---------------------------------------------------------------------
-- Why?
-- Instead of maintaining a separate scoring table for every sport, one
-- common event table records live actions for all six sports.
--
-- Cricket examples:
--   RUN, FOUR, SIX, WIDE, NO_BALL, WICKET
--
-- Football examples:
--   GOAL, OWN_GOAL, YELLOW_CARD, RED_CARD, SUBSTITUTION
--
-- Basketball examples:
--   ONE_POINT, TWO_POINT, THREE_POINT, FOUL, TIMEOUT
--
-- Badminton/Tennis/Volleyball examples:
--   POINT, GAME, SET
--
-- runs/points are numeric so calculations remain easy.
--
-- Important cricket examples:
--   Normal four -> points = 4
--   Wide -> points = 1
--   Wide + four -> points = 5
--   No-ball + four -> points = 5
--
-- For a professional ball-by-ball cricket engine, separate innings/balls
-- tables can be added later. This common table is sufficient for V1 live
-- scoring and is much easier to maintain.
-- ---------------------------------------------------------------------
CREATE TABLE match_events (
    event_id        INT AUTO_INCREMENT PRIMARY KEY,
    match_id        INT NOT NULL,
    team_id         INT NOT NULL,
    player_id       INT NULL,
    event_type      VARCHAR(40) NOT NULL,
    points          INT DEFAULT 0,
    period_no       INT DEFAULT 1,
    event_time      TIME,
    description     VARCHAR(255),
    created_at      DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (match_id) REFERENCES matches(match_id) ON DELETE CASCADE,
    FOREIGN KEY (team_id) REFERENCES teams(team_id),
    FOREIGN KEY (player_id) REFERENCES users(user_id)
);


-- =====================================================================
-- PART 9 : STATISTICS, NOTIFICATIONS & REVIEWS
-- =====================================================================

-- ---------------------------------------------------------------------
-- 9.1 PLAYER STATISTICS
-- ---------------------------------------------------------------------
-- Why?
-- Stores summary statistics used for player profiles and leaderboards.
--
-- The backend can update these values after a match is completed.
-- Keeping a summary table makes leaderboard queries fast and simple.
-- ---------------------------------------------------------------------
CREATE TABLE player_statistics (
    statistic_id   INT AUTO_INCREMENT PRIMARY KEY,
    user_id        INT NOT NULL,
    sport_id       INT NOT NULL,
    matches_played INT DEFAULT 0,
    wins           INT DEFAULT 0,
    losses         INT DEFAULT 0,
    points         INT DEFAULT 0,
    runs           INT DEFAULT 0,
    goals          INT DEFAULT 0,
    created_at     DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at     DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    UNIQUE(user_id, sport_id),
    FOREIGN KEY (user_id) REFERENCES users(user_id) ON DELETE CASCADE,
    FOREIGN KEY (sport_id) REFERENCES sports(sport_id)
);


-- ---------------------------------------------------------------------
-- 9.2 NOTIFICATIONS
-- ---------------------------------------------------------------------
-- Why?
-- Stores messages shown to users inside the application.
--
-- Examples:
--   Opponent accepted your request.
--   Your facility booking is confirmed.
--   Tournament registration is open.
--   Match is starting soon.
-- ---------------------------------------------------------------------
CREATE TABLE notifications (
    notification_id INT AUTO_INCREMENT PRIMARY KEY,
    user_id         INT NOT NULL,
    title           VARCHAR(150) NOT NULL,
    message         VARCHAR(500) NOT NULL,
    notification_type VARCHAR(50),
    is_read         BOOLEAN DEFAULT FALSE,
    created_at      DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(user_id) ON DELETE CASCADE
);


-- ---------------------------------------------------------------------
-- 9.3 REVIEWS
-- ---------------------------------------------------------------------
-- Why?
-- Stores user feedback for facilities.
--
-- A user can review a facility after using it. The application can later
-- calculate average ratings from these rows.
-- ---------------------------------------------------------------------
CREATE TABLE reviews (
    review_id    INT AUTO_INCREMENT PRIMARY KEY,
    facility_id  INT NOT NULL,
    user_id      INT NOT NULL,
    rating       INT NOT NULL,
    review_text  VARCHAR(500),
    created_at   DATETIME DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(facility_id, user_id),
    FOREIGN KEY (facility_id) REFERENCES facilities(facility_id) ON DELETE CASCADE,
    FOREIGN KEY (user_id) REFERENCES users(user_id) ON DELETE CASCADE
);

-- ---------------------------------------------------------------------
-- PROCEDURE 1 : GET NEARBY FACILITIES
-- ---------------------------------------------------------------------
-- Why?
-- Returns approved facilities near a supplied latitude/longitude.
--
-- The application can call this when the user opens "Nearby Facilities".
-- The distance is calculated in kilometres using an Earth-radius value
-- of approximately 6371 km.
-- ---------------------------------------------------------------------
DROP PROCEDURE IF EXISTS sp_nearby_facilities;
DELIMITER $$
CREATE PROCEDURE sp_nearby_facilities(
    IN p_latitude DECIMAL(10,8),
    IN p_longitude DECIMAL(11,8),
    IN p_radius_km DECIMAL(5,2)
)
BEGIN
    SELECT
        f.facility_id,
        f.facility_name,
        s.sport_name,
        f.facility_type,
        f.city,
        f.price_per_hour,
        ROUND(
            6371 * ACOS(
                LEAST(1, GREATEST(-1,
                    COS(RADIANS(p_latitude)) *
                    COS(RADIANS(f.latitude)) *
                    COS(RADIANS(f.longitude) - RADIANS(p_longitude)) +
                    SIN(RADIANS(p_latitude)) *
                    SIN(RADIANS(f.latitude))
                ))
            ), 2
        ) AS distance_km
    FROM facilities f
    JOIN sports s ON f.sport_id = s.sport_id
    WHERE f.approval_status = 'APPROVED'
      AND f.is_active = TRUE
      AND f.latitude IS NOT NULL
      AND f.longitude IS NOT NULL
    HAVING distance_km <= p_radius_km
    ORDER BY distance_km;
END $$
DELIMITER ;


-- ---------------------------------------------------------------------
-- PROCEDURE 2 : FIND OPPONENT TEAMS
-- ---------------------------------------------------------------------
-- Why?
-- Finds active teams playing the requested sport near a given location.
-- The requesting team is excluded.
-- This procedure supports SportsHub's main "Find Opponent" feature.
-- ---------------------------------------------------------------------
DROP PROCEDURE IF EXISTS sp_find_opponents;
DELIMITER $$
CREATE PROCEDURE sp_find_opponents(
    IN p_sport_id INT,
    IN p_team_id INT,
    IN p_latitude DECIMAL(10,8),
    IN p_longitude DECIMAL(11,8),
    IN p_radius_km DECIMAL(5,2)
)
BEGIN
    SELECT
        t.team_id,
        t.team_name,
        s.sport_name,
        t.city,
        ROUND(
            6371 * ACOS(
                LEAST(1, GREATEST(-1,
                    COS(RADIANS(p_latitude)) *
                    COS(RADIANS(t.latitude)) *
                    COS(RADIANS(t.longitude) - RADIANS(p_longitude)) +
                    SIN(RADIANS(p_latitude)) *
                    SIN(RADIANS(t.latitude))
                ))
            ), 2
        ) AS distance_km
    FROM teams t
    JOIN sports s ON t.sport_id = s.sport_id
    WHERE t.sport_id = p_sport_id
      AND t.team_id <> p_team_id
      AND t.is_active = TRUE
      AND t.latitude IS NOT NULL
      AND t.longitude IS NOT NULL
    HAVING distance_km <= p_radius_km
    ORDER BY distance_km;
END $$
DELIMITER ;


-- ---------------------------------------------------------------------
-- PROCEDURE 3 : GET MATCH SCORE
-- ---------------------------------------------------------------------
-- Why?
-- Returns the current stored score and the individual live events for a
-- match. The Flask live-score endpoint can use this information.
-- ---------------------------------------------------------------------
DROP PROCEDURE IF EXISTS sp_match_score;
DELIMITER $$
CREATE PROCEDURE sp_match_score(IN p_match_id INT)
BEGIN
    SELECT
        m.match_id,
        s.sport_name,
        a.team_name AS team_a,
        b.team_name AS team_b,
        m.score_a,
        m.score_b,
        m.status
    FROM matches m
    JOIN sports s ON m.sport_id = s.sport_id
    JOIN teams a ON m.team_a_id = a.team_id
    LEFT JOIN teams b ON m.team_b_id = b.team_id
    WHERE m.match_id = p_match_id;

    SELECT
        event_id,
        team_id,
        player_id,
        event_type,
        points,
        period_no,
        event_time,
        description
    FROM match_events
    WHERE match_id = p_match_id
    ORDER BY event_id;
END $$
DELIMITER ;

-- ---------------------------------------------------------------------
-- TRIGGER 1 : REVIEW RATING VALIDATION
-- ---------------------------------------------------------------------
-- Why?
-- Prevents invalid ratings such as 0 or 7.
-- Only ratings from 1 to 5 are accepted.
-- ---------------------------------------------------------------------
DROP TRIGGER IF EXISTS trg_validate_review_rating;
DELIMITER $$
CREATE TRIGGER trg_validate_review_rating
BEFORE INSERT ON reviews
FOR EACH ROW
BEGIN
    IF NEW.rating < 1 OR NEW.rating > 5 THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Rating must be between 1 and 5';
    END IF;
END $$
DELIMITER ;


-- ---------------------------------------------------------------------
-- TRIGGER 2 : MATCH TEAM VALIDATION
-- ---------------------------------------------------------------------
-- Why?
-- Prevents a team from playing against itself.
-- It also verifies that both teams belong to the sport selected for the
-- match. This protects the most important relationship in the match table.
-- ---------------------------------------------------------------------
DROP TRIGGER IF EXISTS trg_validate_match;
DELIMITER $$
CREATE TRIGGER trg_validate_match
BEFORE INSERT ON matches
FOR EACH ROW
BEGIN
    DECLARE v_sport_a INT;
    DECLARE v_sport_b INT;

    SELECT sport_id INTO v_sport_a
    FROM teams
    WHERE team_id = NEW.team_a_id;

    IF v_sport_a <> NEW.sport_id THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Team A does not belong to selected sport';
    END IF;

    IF NEW.team_b_id IS NOT NULL THEN
        IF NEW.team_b_id = NEW.team_a_id THEN
            SIGNAL SQLSTATE '45000'
            SET MESSAGE_TEXT = 'A team cannot play against itself';
        END IF;

        SELECT sport_id INTO v_sport_b
        FROM teams
        WHERE team_id = NEW.team_b_id;

        IF v_sport_b <> NEW.sport_id THEN
            SIGNAL SQLSTATE '45000'
            SET MESSAGE_TEXT = 'Team B does not belong to selected sport';
        END IF;
    END IF;
END $$
DELIMITER ;


-- ---------------------------------------------------------------------
-- TRIGGER 3 : BOOKING TIME VALIDATION
-- ---------------------------------------------------------------------
-- Why?
-- Prevents an invalid booking where end time is earlier than or equal to
-- start time.
-- ---------------------------------------------------------------------
DROP TRIGGER IF EXISTS trg_validate_booking_time;
DELIMITER $$
CREATE TRIGGER trg_validate_booking_time
BEFORE INSERT ON bookings
FOR EACH ROW
BEGIN
    IF NEW.end_time <= NEW.start_time THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Booking end time must be after start time';
    END IF;
END $$
DELIMITER ;


-- ---------------------------------------------------------------------
-- TRIGGER 4 : SCORE EVENT VALIDATION
-- ---------------------------------------------------------------------
-- Why?
-- Prevents negative points. All scoring values are numeric so totals can
-- be calculated with SUM().
-- ---------------------------------------------------------------------
DROP TRIGGER IF EXISTS trg_validate_match_event;
DELIMITER $$
CREATE TRIGGER trg_validate_match_event
BEFORE INSERT ON match_events
FOR EACH ROW
BEGIN
    IF NEW.points < 0 THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Event points cannot be negative';
    END IF;
END $$
DELIMITER ;

-- ---------------------------------------------------------------------
-- Q17. Find opponent teams near a location.
-- ---------------------------------------------------------------------
-- Why?
-- This is the SQL foundation of the SportsHub Find Opponent feature.
--
-- In simple words:
--   "Find active teams of the selected sport within 10 km."
--
-- The Haversine-style calculation estimates geographic distance.
-- 6371 is the approximate radius of Earth in kilometres.
-- ---------------------------------------------------------------------
SET @my_lat = 17.7200;
SET @my_lon = 83.3000;
SET @sport = 1;
SET @my_team = 1;
SET @radius = 10;

SELECT
    t.team_id,
    t.team_name,
    s.sport_name,
    t.city,
    ROUND(
        6371 * ACOS(
            LEAST(1, GREATEST(-1,
                COS(RADIANS(@my_lat)) *
                COS(RADIANS(t.latitude)) *
                COS(RADIANS(t.longitude) - RADIANS(@my_lon)) +
                SIN(RADIANS(@my_lat)) * SIN(RADIANS(t.latitude))
            ))
        ), 2
    ) AS distance_km
FROM teams t
JOIN sports s ON t.sport_id = s.sport_id
WHERE t.sport_id = @sport
  AND t.team_id <> @my_team
  AND t.is_active = TRUE
HAVING distance_km <= @radius
ORDER BY distance_km;

-- ---------------------------------------------------------------------
-- Q50. Display tournaments near a location.
-- ---------------------------------------------------------------------
-- Why?
-- Uses tournament coordinates to support location-based discovery.
-- ---------------------------------------------------------------------
SET @tour_lat = 17.7000;
SET @tour_lon = 83.2900;

SELECT
    tournament_name,
    city,
    venue,
    ROUND(
        6371 * ACOS(
            LEAST(1, GREATEST(-1,
                COS(RADIANS(@tour_lat)) * COS(RADIANS(latitude)) *
                COS(RADIANS(longitude) - RADIANS(@tour_lon)) +
                SIN(RADIANS(@tour_lat)) * SIN(RADIANS(latitude))
            ))
        ), 2
    ) AS distance_km
FROM tournaments
WHERE status IN ('OPEN','UPCOMING')
HAVING distance_km <= 20
ORDER BY distance_km;

-- ---------------------------------------------------------------------
-- Q60. Find conflicting bookings for a selected time range.
-- ---------------------------------------------------------------------
-- Why?
-- Prevents two users from booking the same facility at overlapping times.
--
-- The overlap rule is:
--   existing_start < new_end
--   AND existing_end > new_start
-- ---------------------------------------------------------------------
SET @facility = 1;
SET @date = '2026-10-26';
SET @new_start = '17:00:00';
SET @new_end = '19:00:00';

SELECT
    booking_id,
    start_time,
    end_time,
    booking_status
FROM bookings
WHERE facility_id = @facility
  AND booking_date = @date
  AND booking_status IN ('PENDING','CONFIRMED')
  AND start_time < @new_end
  AND end_time > @new_start;
